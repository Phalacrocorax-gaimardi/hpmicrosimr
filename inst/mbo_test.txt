#!/bin/bash -l
#SBATCH --job-name=test_mbo
#SBATCH --partition=debug
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=1
#SBATCH --mem=8G
#SBATCH -t 0:10:00

module load R

cat > /tmp/test_mbo.R << 'EOF'
options(mc.cores = 1)
library(mlrMBO)

obj.fun = makeSingleObjectiveFunction(
  fn = function(x) x^2,
  par.set = makeParamSet(makeNumericParam("x", lower = -5, upper = 5))
)

ctrl = makeMBOControl()
ctrl = setMBOControlTermination(ctrl, iters = 2)

# Test different learners
for(learner in c("regr.lm", "regr.km", "regr.randomForest")) {
  cat("Testing", learner, "\n")
  try({
    lrn = makeLearner(learner, predict.type = "response")
    result = mbo(obj.fun, control = ctrl, learner = lrn)
    cat("SUCCESS with", learner, "\n")
  })
}
EOF

Rscript --vanilla /tmp/test_mbo.R
